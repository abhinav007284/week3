✅ 2. Jenkinsfiles (Copy these into your GitHub repo)

Your exam requires correct Jenkinsfile syntax, so here are 3 correct files.

✔ A. Declarative Jenkinsfile (2-Stage Pipeline)
Jenkinsfile
pipeline {
    agent any

    tools {
        maven 'Maven3'      // Must exist in Jenkins → Global Tool Configuration
        jdk 'Java17'        // Example JDK name
    }

    stages {
        stage('Checkout') {
            steps {
                echo "Pulling project from GitHub..."
                checkout scm
            }
        }

        stage('Build') {
            steps {
                echo "Building Maven project..."
                sh 'mvn clean package'
            }
        }
    }

    post {
        success {
            echo "Final Output: BUILD SUCCESS"
        }
        failure {
            echo "Build Failed"
        }
    }
}

✔ B. Declarative Jenkinsfile (3-Stage Pipeline)
Jenkinsfile-3stage
pipeline {
    agent any

    tools {
        maven 'Maven3'
        jdk 'Java17'
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Compile') {
            steps {
                sh 'mvn compile'
            }
        }

        stage('Test') {
            steps {
                sh 'mvn test'
                junit '**/target/surefire-reports/*.xml'
            }
        }

        stage('Package') {
            steps {
                sh 'mvn package'
                archiveArtifacts artifacts: 'target/*.jar', fingerprint: true
            }
        }
    }

    post {
        success {
            echo "Final Output: JAR file generated in target/"
        }
    }
}

✔ C. Scripted Pipeline
Jenkinsfile-scripted
node {
    stage('Checkout') {
        checkout scm
    }

    stage('Build') {
        def mvnHome = tool 'Maven3'
        sh "${mvnHome}/bin/mvn clean package"
    }

    stage('Final') {
        echo "Build completed. Check artifacts."
    }
}
